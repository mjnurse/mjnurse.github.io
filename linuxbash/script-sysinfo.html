sysinfo: Display Linux/macOS system information

#!/usr/bin/env bash
help_text="
usage: sysinfo
-h : This help text.
-a : Show all information.
-d : Show disk usage information.
-n : Show network information.
-p : Show processes information.
-s : Show security information.
"

help_line="Display Linux/macOS system information"
web_desc_line="Display Linux/macOS system information"

set -uo pipefail # Treat unset variables as an error and fail on pipe errors

os_type="$(uname -s)"
is_macos=false
[[ "$os_type" == "Darwin" ]] && is_macos=true

# Terminal Colours
cdef=$'\x1b[39m' # default colour
cbla=$'\x1b[30m'; cgra=$'\x1b[90m'; clgra=$'\x1b[37m'; cwhi=$'\x1b[97m'
cred=$'\x1b[31m'; cgre=$'\x1b[32m'; cyel=$'\x1b[33m'; cblu=$'\x1b[34m'; cmag=$'\x1b[35m'; ccya=$'\x1b[36m';
clred=$'\x1b[91m'; clgre=$'\x1b[92m'; clyel=$'\x1b[93m'; clblu=$'\x1b[94m'; clmag=$'\x1b[95m'; clcya=$'\x1b[96m'

function cecho {
    if [[ "$1" == "-u" ]]; then ul=true; shift; else ul=false; fi
    color=c$1; shift; echo -e "${!color}$*${cdef}"
    if $ul; then echo -e "${!color}${*//?/-}${cdef}"; fi
}

function h1 {
    txt="$*"
    echo
    cecho lmag "$txt"
    cecho lmag "${txt//?/-}"
}

usage() {
    echo "Usage: $0 [options]"
    echo "Options:"
    grep "# arg:" "$0" | grep -v "grep" | sed 's/^ *//; s/).*# *arg://'
    exit 1
}

disk=false
network=false
processes=false
security=false

# Split combined short arguments (eg -abc) into individual arguments (-a -b -c). Remove = symbols.
new_args=()
while [[ "${1:-}" != "" ]]; do
    if [[ ${1:-} == -* && ${1:-} != --* && ${#1} -gt 2 ]]; then
        for ((i=1; i<${#1}; i++)); do
            arg="${1:i:1}"
            if [[ $arg == "=" ]]; then new_args+=("${1:i+1}"); break; fi
            new_args+=("-$arg")
        done
    else
        if [[ ${1:-} == --*=* ]]; then new_args+=("${1%%=*}"); new_args+=("${1##*=}")
        else new_args+=("${1:-}"); fi
    fi
    shift
done

# Process the arguments
set -- ${new_args[@]+"${new_args[@]}"}
while [[ "${1:-}" != "" ]]; do
    case "${1:-}" in
        -a|--all) # arg: Show all information.
            disk=true
            network=true
            processes=true
            security=true
            ;;
        -d|--disk) # arg: Show disk information.
            disk=true
            ;;
        -h|--help) # arg: Show this help text.
            usage
            ;;
        -n|--network) # arg: Show network information.
            network=true
            ;;
        -p|--processes) # arg: Show process information.
            processes=true
            ;;
        -s|--security) # arg: Show security information.
            security=true
            ;;
        -*|--*)
            echo "Unknown option: ${1:-}"
            usage
            ;;
        *)
            break
            ;;
    esac
    shift
done

if $is_macos; then
    os_name="$(sw_vers -productName) $(sw_vers -productVersion)"
    cpu=$(sysctl -n machdep.cpu.brand_string 2>/dev/null || true)
    cores=$(sysctl -n hw.ncpu)

    boot_sec=$(sysctl -n kern.boottime | awk -F'[=,]' '{gsub(/ /, "", $2); print $2}')
    now_sec=$(date +%s)
    up_sec=$((now_sec - boot_sec))
    up_days=$((up_sec / 86400))
    up_hours=$(((up_sec % 86400) / 3600))
    up_mins=$(((up_sec % 3600) / 60))
    up_parts=()
    (( up_days > 0 )) && up_parts+=("$up_days day$( (( up_days != 1 )) && echo s )")
    (( up_hours > 0 )) && up_parts+=("$up_hours hour$( (( up_hours != 1 )) && echo s )")
    (( up_mins > 0 || ${#up_parts[@]} == 0 )) && up_parts+=("$up_mins minute$( (( up_mins != 1 )) && echo s )")
    uptime_str="up $(IFS=,; echo "${up_parts[*]}")"
    uptime_str="${uptime_str//,/, }"

    read -r _ l1 l5 l15 _ <<< "$(sysctl -n vm.loadavg)"

    total_bytes=$(sysctl -n hw.memsize)
    page_size=$(sysctl -n hw.pagesize)
    ram_total=$(awk -v b="$total_bytes" 'BEGIN { g = b / 1073741824; if (g == int(g)) printf "%dGi", g; else printf "%.1fGi", g }')
    ram_detail=$(vm_stat | awk -v total_bytes="$total_bytes" -v ps="$page_size" '
        /Pages wired down:/             { gsub(/\./, "", $NF); wired = $NF }
        /Pages occupied by compressor:/ { gsub(/\./, "", $NF); compressed = $NF }
        /Anonymous pages:/              { gsub(/\./, "", $NF); anon = $NF }
        /Pages purgeable:/              { gsub(/\./, "", $NF); purgeable = $NF }
        /Pages active:/                 { gsub(/\./, "", $NF); active = $NF }
        END {
            total_mb = total_bytes / 1048576
            used_pages = (anon > 0 ? anon - purgeable : active) + wired + compressed
            used_mb = (used_pages * ps) / 1048576
            avail_mb = total_mb - used_mb
            if (avail_mb < 0) avail_mb = 0
            printf "%d MB used, %d MB available of %d MB\n", used_mb, avail_mb, total_mb
        }
    ')
else
    if [[ -r /etc/os-release ]]; then
        . /etc/os-release
        os_name="${PRETTY_NAME:-$(uname -s)}"
    else
        os_name="$(uname -s)"
    fi
    cpu=$(grep -m1 'model name' /proc/cpuinfo 2>/dev/null | cut -d: -f2 | xargs || true)
    cores=$(nproc)
    uptime_str=$(uptime -p)
    read -r l1 l5 l15 _ < /proc/loadavg
    ram_total=$(free -h | awk '/^Mem/ {print $2}')
    ram_detail=$(free -m | awk '/^Mem/ { printf "%d MB used, %d MB available of %d MB\n", $3, $7, $2 }')
fi

h1 "Summary"
cecho lcya "Hostname :${cdef} $(hostname)"
cecho lcya "OS       :${cdef} $os_name"
cecho lcya "Kernel   :${cdef} $(uname -r)"
cecho lcya "Arch     :${cdef} $(uname -m)"
cecho lcya "Uptime   :${cdef} $uptime_str"
cecho lcya "User     :${cdef} $USER (UID $(id -u))"
cecho lcya "Shell    :${cdef} $SHELL"
cecho lcya "CPU      :${cdef} ${cpu:-unknown}"
cecho lcya "Cores    :${cdef} $cores"
cecho lcya "RAM      :${cdef} $ram_total"

mood=$(awk -v l="$l1" -v c="$cores" 'BEGIN { print (l > c) ? "OVERLOADED" : "relaxed" }')
cecho lcya "Load     :${cdef} $l1  $l5  $l15  on $cores cores  ($mood)"
cecho lcya "RAM      :${cdef} $ram_detail"

if $disk; then
    h1 "Disk Usage"
    if $is_macos; then
        df -P -T nodevfs,autofs,tmpfs | awk '
            NR > 1 {
                mount = ""; pct = ""
                for (i = 1; i <= NF; i++) {
                    if (pct == "" && $i ~ /^[0-9]+%$/) {
                        pct = $i
                    } else if (pct != "") {
                        mount = (mount == "" ? $i : mount " " $i)
                    }
                }
                if (pct != "" && mount != "") print mount "\t" pct
            }
        ' | sort
    else
        df -h --output=target,pcent -x tmpfs -x devtmpfs -x squashfs | tail -n +2 |
        awk '{ pct = $NF; $NF = ""; sub(/[[:space:]]+$/, ""); print $0 "\t" pct }' | sort
    fi |
    while IFS=$'\t' read -r mount pct; do
        if [[ "$mount" =~ ^/snap/ ]]; then continue; fi
        n=${pct%\%}
        printf "%-56s %5s  " "$mount" "$pct"
        if   [ "$n" -ge 90 ]; then cecho red "DANGER"
        elif [ "$n" -ge 75 ]; then cecho yel "Watch"
        else                       cecho gre "OK"
        fi
    done
fi

if $processes; then
    h1 "Processes"
    if $is_macos; then
        cecho lgre "Top 5 by CPU"
        ps -Arco pid,user,%cpu,%mem,comm | head -n 6
        echo

        cecho lgre "Top 5 by memory"
        ps -Amco pid,user,%cpu,%mem,comm | head -n 6
        echo

        cecho lcya "Processes running : ${cdef}$(ps -A -o pid= | wc -l | xargs)"
        cecho lcya "Zombie processes  : ${cdef}$(ps -Ao stat= | awk '/^Z/ {c++} END {print c+0}')"
    else
        cecho lgre "Top 5 by CPU"
        ps -eo pid,user,%cpu,%mem,comm --sort=-%cpu | head -n 6
        echo

        cecho lgre "Top 5 by memory"
        ps -eo pid,user,%cpu,%mem,comm --sort=-%mem | head -n 6
        echo

        cecho lcya "Processes running : ${cdef}$(ps -e --no-headers | wc -l)"
        cecho lcya "Zombie processes  : ${cdef}$(ps -eo stat --no-headers | awk '/^Z/ {c++} END {print c+0}')"
    fi
fi

if $network; then
    h1 "Network Information"
    public_ip=$(curl -s --max-time 5 ifconfig.me || true)
    public_ip="${public_ip:-unavailable}"

    if $is_macos; then
        route_info=$(route -n get default 2>/dev/null || true)
        gateway=$(awk '/gateway:/ {print $2; exit}' <<< "$route_info")
        default_if=$(awk '/interface:/ {print $2; exit}' <<< "$route_info")
        local_ip=""
        if [[ -n "$default_if" ]]; then
            local_ip=$(ipconfig getifaddr "$default_if" 2>/dev/null || true)
        fi
        if [[ -z "$local_ip" ]]; then
            local_ip=$(ifconfig | awk '/inet / && $2 != "127.0.0.1" {print $2; exit}')
        fi
        dns=$(scutil --dns 2>/dev/null | awk '/nameserver\[[0-9]+\]/ {print $3}' | awk '!seen[$0]++' | xargs || true)
        if [[ -z "$dns" && -r /etc/resolv.conf ]]; then
            dns=$(awk '/^nameserver/ {print $2}' /etc/resolv.conf | xargs || true)
        fi

        cecho lcya "Local IP  : ${cdef}${local_ip:-unavailable}"
        cecho lcya "Gateway   : ${cdef}${gateway:-unavailable}"
        cecho lcya "DNS       : ${cdef}${dns:-unavailable}"
        cecho lcya "Public IP : ${cdef}$public_ip"
        echo

        cecho lgre "Interfaces:"
        ifconfig | awk '
            function flush_if() {
                if (iface == "") return
                if (addrs == "" && status != "UP") return
                if (status == "") status = (is_up ? "UNKNOWN" : "DOWN")
                printf "%-16s %-14s %s\n", iface, status, addrs
            }
            /^[a-zA-Z0-9._-]+:/ {
                flush_if()
                iface = $1; sub(/:$/, "", iface)
                status = ""; addrs = ""
                is_up = ($0 ~ /<UP[,>]|,UP[,>]/)
                next
            }
            /^[[:space:]]+status:/ {
                if ($2 == "active") status = "UP"
                else status = "DOWN"
            }
            /^[[:space:]]+inet / {
                addrs = (addrs == "" ? $2 : addrs " " $2)
            }
            /^[[:space:]]+inet6 / {
                addr = $2; sub(/%.*/, "", addr)
                addrs = (addrs == "" ? addr : addrs " " addr)
            }
            END { flush_if() }
        ' | sed "
            s/^/${clcya}/;
            s/ / ${cdef}/;
            s/ UNKNOWN / ${cyel}UNKNOWN${cdef} /;
            s/ UP / ${cgre}UP${cdef} /;
            s/ DOWN / ${cred}DOWN${cdef} /;
        "
        echo

        cecho lgre "Ports listening for connections:"
        ports_out=$(
            netstat -an 2>/dev/null | awk '
                function fmt_addr(a,   i) {
                    for (i = length(a); i >= 1; i--) {
                        if (substr(a, i, 1) == ".") return substr(a, 1, i - 1) ":" substr(a, i + 1)
                    }
                    return a
                }
                $1 ~ /^tcp/ && $6 == "LISTEN" { print $1, fmt_addr($4) }
                $1 ~ /^udp/ && $4 != "*.*" && $5 == "*.*" { print $1, fmt_addr($4) }
            ' | sort -u
        )
        if [[ -z "$ports_out" ]] && command -v lsof >/dev/null 2>&1; then
            ports_out=$(
                {
                    lsof -nP -iTCP -sTCP:LISTEN 2>/dev/null | awk 'NR > 1 {print tolower($8), $9}'
                    lsof -nP -iUDP 2>/dev/null | awk 'NR > 1 && $9 != "*:*" && $9 !~ /->/ {print tolower($8), $9}'
                } | sort -u
            )
        fi
        if [[ -n "$ports_out" ]]; then
            printf '%s\n' "$ports_out" | column -t | sed "
                s/^/${clcya}/;
                s/ / ${cdef}/;
            "
        fi
    else
        cecho lcya "Local IP  : ${cdef}$(hostname -I | awk '{print $1}')"
        cecho lcya "Gateway   : ${cdef}$(ip route | awk '/^default/ {print $3; exit}')"
        cecho lcya "DNS       : ${cdef}$(awk '/^nameserver/ {print $2}' /etc/resolv.conf | xargs)"
        cecho lcya "Public IP : ${cdef}$public_ip"
        echo

        cecho lgre "Interfaces:"
        ip -br addr show | sed "
            s/^/${clcya}/;
            s/ / ${cdef}/;
            s/ UNKNOWN / ${cyel}UNKNOWN${cdef} /;
            s/ UP / ${cgre}UP${cdef} /;
            s/ DOWN / ${cred}DOWN${cdef} /;
        "
        echo

        cecho lgre "Ports listening for connections:"
        ss -tuln | awk 'NR > 1 {print $1, $5}' | sort -u | column -t | sed "
            s/^/${clcya}/;
            s/ / ${cdef}/;
        "
    fi
fi

if $security; then
    h1 "Security Information"

    if $is_macos; then
        cecho lcya "Accounts with root power (UID 0):"
        uid0=$(dscl . -list /Users UniqueID 2>/dev/null | awk '$2 == 0 {print "  " $1}' || true)
        if [[ -n "$uid0" ]]; then
            printf '%s\n' "$uid0"
        else
            awk -F: '$3 == 0 {print "  " $1}' /etc/passwd
        fi
        echo

        cecho lcya "Users who can actually log in:"
        login_users=$(dscl . -list /Users UserShell 2>/dev/null | awk '$2 !~ /(nologin|false)$/ {print $1}' | xargs || true)
        if [[ -n "$login_users" ]]; then
            printf '  %s\n' "$login_users"
        else
            grep -Ev '^#|nologin|false$' /etc/passwd | cut -d: -f1 | xargs | sed 's/^/  /'
        fi
        echo

        cecho lcya "Sudo group members (admin):"
        admin_members=$(dscl . -read /Groups/admin GroupMembership 2>/dev/null | sed 's/^GroupMembership: *//' || true)
        printf '  %s\n' "${admin_members:-unavailable}"
        echo

        cecho lcya "Logged in right now:"
        who
        echo

        cecho lcya "Last 5 logins:"
        last -5 | head -n 5
        echo

        fails=$(log show --style syslog --last 24h --predicate 'process == "sshd" AND eventMessage CONTAINS "Failed password"' 2>/dev/null | grep -c "Failed password" || true)
        cecho lcya "Failed SSH password attempts (24h) : ${cdef}$fails"
        cecho lcya "Files in /etc changed today        : ${cdef}$(find /etc/ -type f -mtime -1 2>/dev/null | wc -l | xargs)"
    else
        cecho lcya "Accounts with root power (UID 0):"
        awk -F: '$3 == 0 {print "  " $1}' /etc/passwd
        echo

        cecho lcya "Users who can actually log in:"
        grep -Ev 'nologin|false$' /etc/passwd | cut -d: -f1 | xargs | sed 's/^/  /'
        echo

        cecho lcya "Sudo group members:"
        getent group sudo wheel | cut -d: -f4 | tr ',' ' ' | xargs | sed 's/^/  /'
        echo

        cecho lcya "Logged in right now:"
        who
        echo

        cecho lcya "Last 5 logins:"
        last -n 5 | head -n 5
        echo

        if [ -r /var/log/auth.log ]; then
            fails=$(grep -c "Failed password" /var/log/auth.log || true)
        else
            fails=$(journalctl -q -u ssh -u sshd 2>/dev/null | grep -c "Failed password" || true)
        fi
        cecho lcya "Failed SSH password attempts : ${cdef}$fails"
        cecho lcya "Files in /etc changed today  : ${cdef}$(find /etc/ -type f -mtime -1 2>/dev/null | wc -l | xargs)"
    fi
fi

if ! $processes || ! $network || ! $disk || ! $security; then
    echo
    cecho gra "Available options:"
    if ! $processes; then
        cecho gra "-p : Show processes information"
    fi
    if ! $network; then
        cecho gra "-n : Show network information"
    fi
    if ! $disk; then
        cecho gra "-d : Show disk usage information"
    fi
    if ! $security; then
        cecho gra "-s : Show security information"
    fi
fi
