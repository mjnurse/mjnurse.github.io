clr: Check Local Repositories for uncommitted changes

#!/usr/bin/env bash

help_text="
NAME
    clr - Check Local Repositories for uncommitted changes.

USAGE
    clr [options]

OPTIONS
    -p|--push
        Interactively commit, merge remote changes, and push for each repository
        with uncommitted, unpushed, or unpulled changes.

    -f|--force
        With -p, commit (with default message 'Various'), merge, and push without
        prompting for confirmation unless a conflict occurs.

    -h|--help
        Show help text.

DESCRIPTION
    Checks a predefined list of local git repositories for uncommitted
    changes, unpushed/unpulled commits, or unfinished rebases/merges. When run
    without options, displays a summary of repositories needing attention.
    With the -p option, iterates through each repository and prompts to
    commit, pull (merge), and push. Works on both macOS and Ubuntu WSL.

AUTHOR
    mjnurse.github.io - 2026
"
help_line="Check Local Repositories for uncommitted changes"
web_desc_line="Check Local Repositories for uncommitted changes"

# set -x

# Terminal Colours (POSIX-compatible \033 escapes for macOS and Linux/WSL)
cdef="\033[39m" # default colour
cbla="\033[30m"; cgra="\033[90m"; clgra="\033[37m"; cwhi="\033[97m"
cred="\033[31m"; cgre="\033[32m"; cyel="\033[33m"
cblu="\033[34m"; cmag="\033[35m"; ccya="\033[36m"
clred="\033[91m"; clgre="\033[92m"; clyel="\033[93m"
clblu="\033[94m"; clmag="\033[95m"; clcya="\033[96m"

function cecho {
    local color="c$1"; shift
    printf '%b\n' "${!color}$*${cdef}"
}

function to_upper {
    printf '%s' "$1" | tr '[:lower:]' '[:upper:]'
}

try="Try ${0##*/} -h for more information"
tmp="${help_text##*USAGE}"
usage=$(echo "Usage: ${tmp%%OPTIONS*}" | tr -d "\n" | sed "s/  */ /g")

push_yn=n
force_yn=n
while [[ "$1" != "" ]]; do
    case $1 in
        -h|--help)
            echo "$help_text"
            exit 0
            ;;
        -p|--push)
            push_yn=y
            ;;
        -f|--force)
            force_yn=y
            ;;
        ?*)
            cecho lred "Unknown option: $1"
            echo "$usage"
            echo "$try"
            exit 1
            ;;
    esac
    shift
done

repos=(
    "$HOME/mjnurse/bash"
    "$HOME/mjnurse/python"
    "$HOME/mjnurse/cli-builder"
    "$HOME/mjnurse/website-builder"
    "$HOME/mjnurse/mjnurse-website"
    "$HOME/mjnurse/mjnurse.github.io"
)

function is_rebase_in_progress {
    local git_dir
    git_dir="$(git rev-parse --git-dir 2>/dev/null)" || return 1
    [[ -d "$git_dir/rebase-merge" || -d "$git_dir/rebase-apply" ]]
}

function is_merge_in_progress {
    local git_dir
    git_dir="$(git rev-parse --git-dir 2>/dev/null)" || return 1
    [[ -f "$git_dir/MERGE_HEAD" ]]
}

function has_unmerged_conflicts {
    [[ -n "$(git diff --name-only --diff-filter=U 2>/dev/null)" ]]
}

function find_conflict_marker_files {
    local status_line xy file
    while IFS= read -r status_line; do
        [[ -z "$status_line" ]] && continue
        xy="${status_line:0:2}"
        file="${status_line:3}"
        if [[ "$xy" == R* || "$xy" == C* ]]; then
            file="${file#* -> }"
        fi
        if [[ -f "$file" && ! -L "$file" ]]; then
            if grep -qE '^<<<<<<< ' "$file" 2>/dev/null && grep -qE '^>>>>>>> ' "$file" 2>/dev/null; then
                printf '%s\n' "$file"
            fi
        fi
    done < <(git -c core.quotepath=false status --porcelain)
}

function get_branch {
    git symbolic-ref --short -q HEAD 2>/dev/null
}

function get_ahead_behind {
    # Sets global ahead_count, behind_count, and upstream_ref
    ahead_count=0
    behind_count=0
    upstream_ref=""

    if git rev-parse --abbrev-ref --symbolic-full-name @{u} >/dev/null 2>&1; then
        upstream_ref="@{u}"
    else
        local branch
        branch="$(get_branch)"
        if [[ -n "$branch" ]] && git rev-parse --verify "origin/$branch" >/dev/null 2>&1; then
            upstream_ref="origin/$branch"
        fi
    fi

    if [[ -n "$upstream_ref" ]]; then
        local counts
        counts="$(git rev-list --left-right --count "${upstream_ref}...HEAD" 2>/dev/null)"
        if [[ -n "$counts" ]]; then
            read -r behind_count ahead_count <<< "$counts"
        fi
    fi
}

function resolve_merge_conflicts {
    local side="$1" # "ours" (local) or "theirs" (remote)
    local status_line xy file
    while IFS= read -r status_line; do
        [[ -z "$status_line" ]] && continue
        xy="${status_line:0:2}"
        file="${status_line:3}"
        case "$xy" in
            UU|AA|DD|AU|UA|DU|UD)
                if [[ "$side" == "ours" ]]; then
                    if [[ "$xy" == "DU" || "$xy" == "DD" ]]; then
                        git rm -f -- "$file"
                    else
                        git checkout --ours -- "$file" && git add -- "$file"
                    fi
                else
                    if [[ "$xy" == "UD" || "$xy" == "DD" ]]; then
                        git rm -f -- "$file"
                    else
                        git checkout --theirs -- "$file" && git add -- "$file"
                    fi
                fi
                ;;
        esac
    done < <(git -c core.quotepath=false status --porcelain)
}

function ensure_ssh_remote {
    # Switch GitHub HTTPS origin to SSH so pushing works consistently across Mac and WSL
    local remote_url
    remote_url="$(git remote get-url origin 2>/dev/null)"
    if [[ "$remote_url" =~ ^https://github\.com/(.+)$ ]]; then
        local ssh_url="git@github.com:${BASH_REMATCH[1]}"
        [[ "$ssh_url" != *.git ]] && ssh_url="${ssh_url}.git"
        cecho lyel "Note: origin uses HTTPS ($remote_url). Switching to SSH ($ssh_url)..."
        git remote set-url origin "$ssh_url"
    fi
}

function merge_and_push {
    local repo_dir="$1"
    local branch
    branch="$(get_branch)"

    ensure_ssh_remote
    get_ahead_behind

    if (( behind_count > 0 )); then
        cecho lcya "Merging $behind_count remote commit(s)..."
        local pull_ok=n
        if [[ "$upstream_ref" == "@{u}" ]]; then
            git pull --no-rebase --no-edit && pull_ok=y
        elif [[ -n "$branch" ]]; then
            git pull --no-rebase --no-edit origin "$branch" && pull_ok=y
        fi

        if [[ "$pull_ok" == "n" ]]; then
            if is_merge_in_progress; then
                cecho lred "Merge conflict encountered in $repo_dir!"
                git status --short
                local res=""
                local res_up=""
                while [[ "$res_up" != "L" && "$res_up" != "R" && "$res_up" != "M" && "$res_up" != "A" ]]; do
                    echo
                    printf "%b" "${clyel}Resolve conflict using [L]ocal (this machine), [R]emote (GitHub), [M]anual, or [A]bort merge [lrmA]? ${cdef}"
                    read -r res
                    res="${res:-A}"
                    res_up="$(to_upper "$res")"
                done
                case "$res_up" in
                    L)
                        resolve_merge_conflicts ours
                        if ! git commit --no-edit; then
                            cecho lred "Failed to commit merge resolution. Aborting merge."
                            git merge --abort
                            return 1
                        fi
                        cecho lgre "Merge conflict resolved using local version."
                        ;;
                    R)
                        resolve_merge_conflicts theirs
                        if ! git commit --no-edit; then
                            cecho lred "Failed to commit merge resolution. Aborting merge."
                            git merge --abort
                            return 1
                        fi
                        cecho lgre "Merge conflict resolved using remote version."
                        ;;
                    M)
                        cecho lyel "Leaving merge conflict in $repo_dir for manual resolution. Re-run 'clr -p' once resolved."
                        return 1
                        ;;
                    A)
                        git merge --abort
                        cecho lyel "Merge aborted for $repo_dir (local commits preserved on branch)."
                        return 1
                        ;;
                esac
            else
                cecho lred "Failed to pull changes in $repo_dir."
                return 1
            fi
        fi
    fi

    get_ahead_behind
    if (( ahead_count > 0 )) || [[ -z "$upstream_ref" ]]; then
        cecho lcya "Pushing to remote..."
        local push_ok=n
        if [[ "$upstream_ref" == "@{u}" ]]; then
            git push && push_ok=y
        elif [[ -n "$branch" ]]; then
            git push -u origin "$branch" && push_ok=y
        fi

        if [[ "$push_ok" == "y" ]]; then
            cecho lgre "Successfully pushed $repo_dir."
        else
            cecho lred "Failed to push $repo_dir!"
            return 1
        fi
    else
        cecho lgre "$repo_dir is up to date with remote."
    fi
}

echo
cecho lgre "Checking local git repositories for uncommitted changes..."

if [[ "$push_yn" == "n" ]]; then
    changes_yn=n
    for d in "${repos[@]}"; do
        if [[ -d "$d" ]]; then
            cd "$d" || continue
            if is_rebase_in_progress || is_merge_in_progress; then
                cecho lred "Rebase/merge in progress in repo: ${clcya}$d"
                changes_yn=y
            elif has_unmerged_conflicts; then
                cecho lred "Unresolved conflicts (e.g. from stash pop) in repo: ${clcya}$d"
                changes_yn=y
            elif ! git symbolic-ref -q HEAD >/dev/null 2>&1; then
                cecho lred "Detached HEAD in repo: ${clcya}$d"
                changes_yn=y
            fi
            if [[ -n "$(find_conflict_marker_files)" ]]; then
                cecho lred "Conflict markers found in modified files in repo: ${clcya}$d"
                changes_yn=y
            fi
            if [[ -n "$(git status --porcelain)" ]]; then
                cecho cya "Uncommitted changes in repo: ${clcya}$d"
                changes_yn=y
            fi
            get_ahead_behind
            if (( ahead_count > 0 && behind_count > 0 )); then
                cecho lyel "Diverged ($ahead_count ahead, $behind_count behind) in repo: ${clcya}$d"
                changes_yn=y
            elif (( ahead_count > 0 )); then
                cecho lyel "Unpushed commits ($ahead_count ahead) in repo: ${clcya}$d"
                changes_yn=y
            elif (( behind_count > 0 )); then
                cecho lyel "Unpulled remote commits ($behind_count behind) in repo: ${clcya}$d"
                changes_yn=y
            fi
        fi
    done
    if [[ "$changes_yn" == "y" ]]; then
        echo
        cecho lmag "Run clr -p to selectively commit/merge/push changes."
        exit 1
    fi
    exit 0
fi

for d in "${repos[@]}"; do
    if [[ -d "$d" ]]; then
        cd "$d" || continue
        echo "Checking repository: $d"
        if ! git fetch --quiet; then
            cecho lred "Failed to fetch remote for $d (check network/auth). Skipping."
            continue
        fi

        # Handle unfinished rebase first
        if is_rebase_in_progress; then
            echo
            cecho lred "------------------------------------------------------------"
            cecho lred "Repository: $d (REBASE IN PROGRESS)"
            cecho lred "------------------------------------------------------------"
            git status
            echo
            printf "%b" "${clyel}Continue rebase (C), abort rebase (A), or skip (S) [caS]? ${cdef}"
            read -r rb_choice
            rb_choice="$(to_upper "${rb_choice:-S}")"
            if [[ "$rb_choice" == "A" ]]; then
                git rebase --abort
                cecho lyel "Rebase aborted."
            elif [[ "$rb_choice" == "C" ]]; then
                git add -A
                if GIT_EDITOR=true git rebase --continue; then
                    cecho lgre "Rebase completed."
                    merge_and_push "$d"
                else
                    cecho lred "Could not complete rebase automatically."
                fi
                continue
            else
                continue
            fi
        fi

        # Handle unfinished merge
        if is_merge_in_progress; then
            echo
            cecho lred "------------------------------------------------------------"
            cecho lred "Repository: $d (MERGE IN PROGRESS)"
            cecho lred "------------------------------------------------------------"
            git status
            echo
            printf "%b" "${clyel}Continue merge (C), use [L]ocal, use [R]emote, [A]bort merge, or [S]kip [clraS]? ${cdef}"
            read -r mg_choice
            mg_choice="$(to_upper "${mg_choice:-S}")"
            case "$mg_choice" in
                A)
                    git merge --abort
                    cecho lyel "Merge aborted."
                    ;;
                L)
                    resolve_merge_conflicts ours
                    if git commit --no-edit; then
                        cecho lgre "Merge completed using local version."
                        merge_and_push "$d"
                    else
                        cecho lred "Could not complete merge automatically."
                    fi
                    continue
                    ;;
                R)
                    resolve_merge_conflicts theirs
                    if git commit --no-edit; then
                        cecho lgre "Merge completed using remote version."
                        merge_and_push "$d"
                    else
                        cecho lred "Could not complete merge automatically."
                    fi
                    continue
                    ;;
                C)
                    if [[ -n "$(git diff --name-only --diff-filter=U)" ]]; then
                        cecho lred "Unresolved conflicts still exist in $d. Resolve them first or choose L/R/A."
                    elif git commit --no-edit; then
                        cecho lgre "Merge completed."
                        merge_and_push "$d"
                    else
                        cecho lred "Could not complete merge."
                    fi
                    continue
                    ;;
                *)
                    continue
                    ;;
            esac
        fi

        # Handle unmerged conflicts outside a merge/rebase (e.g. from 'git stash pop')
        if has_unmerged_conflicts; then
            echo
            cecho lred "------------------------------------------------------------"
            cecho lred "Repository: $d (UNRESOLVED CONFLICTS)"
            cecho lred "------------------------------------------------------------"
            cecho lyel "Unmerged files detected (e.g. from 'git stash pop' or cherry-pick):"
            git diff --name-only --diff-filter=U
            echo
            if [[ "$force_yn" == "y" ]]; then
                cecho lred "Skipping $d (cannot auto-commit while unresolved conflicts exist)."
                continue
            fi
            printf "%b" "${clyel}Resolve using [U]pstream/HEAD (ours), [T]heirs (stashed/incoming), or [S]kip to resolve manually [utS]? ${cdef}"
            read -r cf_choice
            cf_choice="$(to_upper "${cf_choice:-S}")"
            case "$cf_choice" in
                U)
                    resolve_merge_conflicts ours
                    cecho lgre "Conflicts resolved using HEAD/upstream version. Review and re-run 'clr -p' to commit."
                    ;;
                T)
                    resolve_merge_conflicts theirs
                    cecho lgre "Conflicts resolved using incoming/stashed version. Review and re-run 'clr -p' to commit."
                    ;;
                *)
                    cecho lyel "Skipping $d. Resolve conflicts manually, then re-run 'clr -p'."
                    ;;
            esac
            continue
        fi

        # Guard against detached HEAD
        if ! git symbolic-ref -q HEAD >/dev/null 2>&1; then
            cecho lred "Repository $d is in detached HEAD state. Checkout a branch (e.g. 'git checkout main') first. Skipping."
            continue
        fi

        has_uncommitted=n
        [[ -n "$(git status --porcelain)" ]] && has_uncommitted=y
        get_ahead_behind

        if [[ "$has_uncommitted" == "y" ]]; then
            conflict_files="$(find_conflict_marker_files)"
            if [[ -n "$conflict_files" ]]; then
                echo
                cecho lred "------------------------------------------------------------"
                cecho lred "Repository: $d (CONFLICT MARKERS DETECTED)"
                cecho lred "------------------------------------------------------------"
                cecho lred "Git conflict markers (<<<<<<< / >>>>>>>) found in modified file(s):"
                printf '%s\n' "$conflict_files"
                if [[ "$force_yn" == "y" ]]; then
                    cecho lred "Refusing to auto-commit $d while conflict markers are present. Skipping."
                    continue
                fi
                echo
                printf "%b" "${clyel}Conflict markers detected. Still proceed to commit prompt [yN]? ${cdef}"
                read -r cm_override
                if [[ "$(to_upper "${cm_override:-N}")" != "Y" ]]; then
                    cecho lyel "Skipping $d. Resolve conflict markers first, then re-run 'clr -p'."
                    continue
                fi
            fi

            echo
            cecho lmag "------------------------------------------------------------"
            cecho lmag "Repository: $d"
            cecho lmag "------------------------------------------------------------"
            git status

            if [[ "$force_yn" == "y" ]]; then
                yn="Y"
                cm="Various"
            else
                yn=""
            fi

            yn_up="$(to_upper "$yn")"
            while [[ "$yn_up" != "Y" && "$yn_up" != "N" ]]; do
                echo
                printf "%b" "${clcya}Commit and Sync changes with comment Various (YN), custom comment (C) or view diff (D) [cdyN]? ${cdef}"
                read -r yn
                yn="${yn:-N}"
                yn_up="$(to_upper "$yn")"

                case "$yn_up" in
                    D)
                        git diff
                        ;;
                    C)
                        printf "%b" "${clcya}Commit Message (blank defaults to 'Various'): ${cdef}"
                        read -r cm
                        cm="${cm:-Various}"
                        yn_up="Y"
                        ;;
                    Y)
                        cm="Various"
                        printf "%b\n" "${clcya}Using default commit message:${cdef} Various"
                        ;;
                    N)
                        ;;
                    *)
                        cecho lyel "Invalid choice: $yn"
                        ;;
                esac
            done

            if [[ "$yn_up" == "Y" ]]; then
                git add -A
                if git commit -m "$cm"; then
                    merge_and_push "$d"
                else
                    cecho lred "Commit failed in $d."
                fi
            fi
        elif (( ahead_count > 0 || behind_count > 0 )); then
            echo
            cecho lmag "------------------------------------------------------------"
            cecho lmag "Repository: $d (clean working tree; ahead $ahead_count, behind $behind_count)"
            cecho lmag "------------------------------------------------------------"
            git status -sb
            if [[ "$force_yn" == "y" ]]; then
                sync_yn="Y"
            else
                echo
                printf "%b" "${clcya}Sync (merge/push) this repository with origin [yN]? ${cdef}"
                read -r sync_yn
                sync_yn="$(to_upper "${sync_yn:-N}")"
            fi
            if [[ "$sync_yn" == "Y" ]]; then
                merge_and_push "$d"
            fi
        fi
    fi
done
